-- A simple, workable AST for Python source code.

Identifier
* Value String

Object

Singleton

Position
* Line Integer
* ColOffset Integer

Node :: Object
* Position Position

ExpressionContext = {Load, Store, Del, AugLoad, AugStore, Param}

BoolOp = {And, Or}

Op = {Add, Sub, Mult, Div, Modulo, Pow, LShift, RShift, BitOr, BitXor, BitAnd, FloorDiv}

UnaryOp = {Invert, Not, UAdd, USub}

BinaryOp = {Eq, NotEq, Lt, LtE, Gt, GtE, Is, IsNot, In, NotIn}

AST :: Node
* Nodes []Node
* Add (Node Node)

Module :: Node
* Body []Statement

Expression :: Node
* Body []Statement
* ExpressionType String

Suite :: Node
* Body []Statement

Statement :: Node
* StatementType String

Bytes :: Node

DocString :: Expression

CommentLine :: Statement
* Text String

FunctionDef :: Statement
* Name Identifier
* Args Arguments
* Body []Statement
* DecoratorList []Expression
* Returns Expression -- The return type. The return expression in the body is somethign ddifferent.

ClassDef :: Statement
* Name Identifier
* Bases []Expression
* Keywords []KeywordNode
* Starargs Expression
* Kwargs Expression
* Body []Statement
* DecoratorList []Expression

Keyword = {False, None, True, _and, _as, _assert, _async, _await, _break, _class, _continue, _def, _del, _elif, _else, _except, _finally, _for, _from, _global, _if, _import, _in, _is, _lambda, _nonlocal, _not, _or, _pass, _raise, _return, _try, _while, _with, _yield}
